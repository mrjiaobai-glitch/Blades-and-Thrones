blades_and_thrones_junta.1 = { #军政府支持事件：花钱/稳定/合法性换掌控力
	type = country_event
	title = blades_and_thrones_junta.1.title
	desc = blades_and_thrones_junta.1.desc
	outcome = neutral
	trigger = {
	}

	immediate = {
		hidden_effect = {
			# 军政府稳定度变量初始化（控制台直发/灾难外测试时变量不存在，change_variable 会静默失效）
			if = {
				limit = { NOT = { has_variable = blades_and_thrones_junta_stability } }
				set_variable = {
					name = blades_and_thrones_junta_stability
					value = 50
				}
			}
			# 存入本次"让利士兵"的金额（6个月军饷），供选项 tooltip 动态显示
			set_variable = {
				name = blades_and_thrones_junta_handout_display
				value = blades_and_thrones_junta_soldier_handout_cost
			}
		}
	}

	option = {
		name = blades_and_thrones_junta.1.a
		ai_chance = {
			factor = 10
			modifier = {
				factor = 5
				gold >= {
					value = root.blades_and_thrones_junta_soldier_handout_cost
					multiply = 2
				}
			}
		}
		trigger = {
			gold >= {
				value = root.blades_and_thrones_junta_soldier_handout_cost
			}
		}
		add_gold = {
			value = root.blades_and_thrones_junta_soldier_handout_cost
			multiply = -1
		}
		"estate(estate_type:zzz_rebalanced_nepal_mercenary_soldiers_estate)" = {
			estate_add_gold = {
				value = root.blades_and_thrones_junta_soldier_handout_cost
			}
		}
		add_estate_satisfaction = {
			type = estate_type:zzz_rebalanced_nepal_mercenary_soldiers_estate
			value = estate_satisfaction_mild_bonus
		}
		change_variable = {
			name = blades_and_thrones_junta_stability
			add = 8
			min = 0
			max = 100
		}
		custom_tooltip = blades_and_thrones_junta.1.a.tt
	}

	option = {
		name = blades_and_thrones_junta.1.b
		ai_chance = {
			factor = 10
			modifier = {
				factor = 5
				gold >= {
					value = root.\
					multiply = 2
				}
			}
		}
		trigger = {
			gold >= {
				value = root.blades_and_thrones_junta_estate_bribe_cost
			}
		}
		add_gold = {
			value = root.blades_and_thrones_junta_estate_bribe_cost
			multiply = -1
		}
		add_legitimacy = legitimacy_mild_bonus
		add_estate_satisfaction = {
			type = estate_type:nobles_estate
			value = estate_satisfaction_weak_bonus
		}
		add_estate_satisfaction = {
			type = estate_type:burghers_estate
			value = estate_satisfaction_weak_bonus
		}
		add_estate_satisfaction = {
			type = estate_type:zzz_rebalanced_nepal_mercenary_soldiers_estate
			value = estate_satisfaction_weak_penalty
		}
		change_variable = {
			name = blades_and_thrones_junta_stability
			add = 6
			min = 0
			max = 100
		}
		custom_tooltip = blades_and_thrones_junta.1.b.tt
	}

	option = {
		name = blades_and_thrones_junta.1.c
		ai_chance = {
			factor = 35
		}
		add_stability = stability_mild_penalty
		add_legitimacy = legitimacy_weak_penalty
		add_estate_satisfaction = {
			type = estate_type:zzz_rebalanced_nepal_mercenary_soldiers_estate
			value = estate_satisfaction_mild_bonus
		}
		change_variable = {
			name = blades_and_thrones_junta_stability
			add = 5
			min = 0
			max = 100
		}
		custom_tooltip = blades_and_thrones_junta.1.c.tt
	}

	option = {
		name = blades_and_thrones_junta.1.d
		ai_chance = {
			factor = 10
			modifier = {
		ai_chance = {
			factor = 10
			modifier = {
				factor = 5
				gold >= {
					value = root.\
					multiply = 2
				}
			}
		}
			}
		}
		add_gold = {
			value = root.blades_and_thrones_junta_admin_cost
			multiply = -1
		}
		add_legitimacy = legitimacy_mild_bonus
		add_stability = stability_weak_bonus
		change_variable = {
			name = blades_and_thrones_junta_stability
			add = 6
			min = 0
			max = 100
		}
		custom_tooltip = blades_and_thrones_junta.1.d.tt
	}
}
