mercenary_mutiny_test.2 = { #切换扮演：接管已生成的佣兵军团国
	type = country_event
	title = mercenary_mutiny_test.2.title
	desc = mercenary_mutiny_test.2.desc
	outcome = neutral
	trigger = {
		is_human = yes
	}

	immediate = {
		error_log = "MERC_DEBUG .2 fired"
		var:mercenary_mutiny_army_country_ref = {
			save_scope_as = mercenary_mutiny_army_country
		}
		if = {
			limit = {
				has_variable = mercenary_mutiny_annexer_country_ref
			}
			var:mercenary_mutiny_annexer_country_ref = {
				save_scope_as = mercenary_mutiny_annexer_country
			}
		}
	}

	option = {
		name = mercenary_mutiny_test.2.a
		high_risk_option = yes
		trigger = {
			is_human = yes
		}
		error_log = "MERC_DEBUG .2 option A selected"
		ROOT = {
			every_character = {
				limit = {
					is_alive = yes
					NOT = { owner ?= scope:mercenary_mutiny_army_country }
					NOT = {
						OR = {
							is_ruler = yes
							is_regent = yes
							is_heir = yes
						}
					}
					OR = {
						is_general = yes
						is_admiral = yes
						has_unit = yes
						AND = {
							exists = scope:mercenary_mutiny_annexer_country
							is_courtier = yes
							culture ?= {
								NOR = {
									is_primary_or_accepted_in = scope:mercenary_mutiny_annexer_country
									is_tolerated_in = scope:mercenary_mutiny_annexer_country
								}
							}
						}
					}
				}
				move_country = scope:mercenary_mutiny_army_country
				remove_variable = mercenary_mutiny_exiled_candidate
			}
		}
		scope:mercenary_mutiny_army_country = {
			force_recalc_country_active_status = scope:mercenary_mutiny_army_country
			change_country_type = army
		}
		error_log = "MERC_DEBUG .2 before change_player"
		change_player = scope:mercenary_mutiny_army_country
		error_log = "MERC_DEBUG .2 after change_player"
	}

	option = {
		name = mercenary_mutiny_test.2.b
		custom_tooltip = civil_war_game_over
	}
}

mercenary_mutiny_test.3 = { #吞并转军团测试：模拟"国家被吞并→残部转流亡者军团国"
	# 用法（两种，按触发者身份区分）：
	# · event mercenary_mutiny_test.3 <AI国TAG>：在 AI 吞并国上触发。
	#   immediate 的 AI 路径随机选一名人类玩家作为被吞并对象，对其执行
	#   转换并弹出"国家覆灭"（test.2）选择事件；触发者（AI）自己不会被选中。
	# · event mercenary_mutiny_test.3（玩家在自己身上触发）：
	#   immediate 不动作，玩家直接看到本事件，选"转换并直接切换玩家"，
	#   把自己当被吞并对象转换并切换扮演（不走 test.2）。
	type = country_event
	title = mercenary_mutiny_test.3.title
	desc = mercenary_mutiny_test.3.desc
	outcome = neutral
	orphan = yes
	trigger = {
		always = no
	}

	immediate = {
		hidden_effect = {
			# AI 路径：仅在 AI 触发时执行——随机选一名人类玩家（排除触发者自己），
			# 对其执行"被吞并→流亡者"转换，然后向该玩家弹出"国家覆灭"选择事件。
			if = {
				limit = {
					is_ai = yes
				}
				error_log = "MERC_DEBUG .3 immediate started"
				random_country = {
					limit = {
						is_human = yes
						NOT = { this = prev }
					}
					save_scope_as = mercenary_mutiny_annex_target
				}
				scope:mercenary_mutiny_annex_target = {
					mercenary_mutiny_convert_annexed_to_army_country_effect = yes
					error_log = "MERC_DEBUG .3 conversion done"
					if = {
						limit = {
							has_variable = mercenary_mutiny_army_country_ref
						}
						var:mercenary_mutiny_army_country_ref = {
							save_scope_as = mercenary_mutiny_army_country
						}
						error_log = "MERC_DEBUG .3 triggering .2"
						trigger_event_non_silently = {
							id = mercenary_mutiny_test.2
						}
					}
				}
			}
		}
	}

	option = {
		name = mercenary_mutiny_test.3.a
		trigger = {
			is_ai = yes
		}
		ai_chance = 1
		custom_tooltip = mercenary_mutiny_test.3.a.tt
	}

	option = {
		name = mercenary_mutiny_test.3.b
		high_risk_option = yes
		trigger = {
			is_human = yes
		}
		custom_tooltip = mercenary_mutiny_test.3.b.tt
		error_log = "MERC_DEBUG .3 direct option started"
		mercenary_mutiny_convert_annexed_to_army_country_effect = yes
		var:mercenary_mutiny_army_country_ref = {
			save_scope_as = mercenary_mutiny_army_country
		}
		scope:mercenary_mutiny_army_country = {
			force_recalc_country_active_status = scope:mercenary_mutiny_army_country
			change_country_type = army
		}
		error_log = "MERC_DEBUG .3 before direct change_player"
		change_player = scope:mercenary_mutiny_army_country
		error_log = "MERC_DEBUG .3 after direct change_player"
	}
}

# 佣兵军团国（不忠诚附属国）索饷事件——由月度脉冲触发，也可用调试菜单选项 j 触发
mercenary_mutiny_test.10 = { #军团索饷：不忠附属国的巨额军饷要求
	type = country_event
	title = mercenary_mutiny_test.10.title
	desc = mercenary_mutiny_test.10.desc
	outcome = negative

	immediate = {
		hidden_effect = {
			# 定位索饷的军团国：优先用宗主保存的军团引用（外派/夺舍两条路径都设置
			# mercenary_mutiny_company_ref），失败时兜底按不忠标记随机找。
			# 教训：random_subject + has_variable 组合实测不可靠（vanilla 无此先例），
			# 症状是索饷事件弹出但付钱/拒绝全部无效、独立事件弹出却不独立。
			if = {
				limit = {
					exists = var:mercenary_mutiny_company_ref
				}
				var:mercenary_mutiny_company_ref = {
					save_scope_as = mercenary_mutiny_company_scope
				}
			}
			if = {
				limit = {
					NOT = { exists = scope:mercenary_mutiny_company_scope }
				}
				random_subject = {
					limit = {
						has_variable = mercenary_mutiny_disloyal_subject
						NOT = { has_country_modifier = mercenary_mutiny_placated }
					}
					save_scope_as = mercenary_mutiny_company_scope
				}
			}
			# 存入本次索饷金额，供选项 tooltip 动态显示
			set_variable = {
				name = blades_and_thrones_company_demand_display
				value = mercenary_mutiny_company_demand_cost
			}
			if = {
				limit = {
					NOT = { has_variable = blades_and_thrones_military_crisis }
				}
				set_variable = {
					name = blades_and_thrones_military_crisis
					value = 0
				}
			}
		}
	}

	option = {
		name = mercenary_mutiny_test.10.a
		trigger = {
			gold >= {
				value = root.mercenary_mutiny_company_demand_cost
			}
		}
		# AI 宗主决策（2024-08 放开 AI 后）：财政宽裕就付钱安抚，拮据/危机深重就拒绝
		ai_chance = {
			factor = 60
			modifier = {
				factor = 1.5
				gold >= {
					value = root.mercenary_mutiny_company_demand_cost
					multiply = 3
				}
			}
			modifier = {
				factor = 0.5
				has_variable = blades_and_thrones_military_crisis
				var:blades_and_thrones_military_crisis >= 60
			}
		}
		custom_tooltip = mercenary_mutiny_test.10.a.tt
		hidden_effect = {
			if = {
				limit = {
					exists = scope:mercenary_mutiny_company_scope
				}
				add_gold = {
					value = root.mercenary_mutiny_company_demand_cost
					multiply = -1
				}
				scope:mercenary_mutiny_company_scope = {
					add_country_modifier = {
						modifier = mercenary_mutiny_placated
						years = 5
						mode = add_and_extend
					}
				}
				change_variable = {
					name = blades_and_thrones_military_crisis
					add = -10
					min = 0
					max = 100
				}
				set_variable = {
					name = blades_and_thrones_military_crisis_display
					value = var:blades_and_thrones_military_crisis
				}
			}
		}
	}

	option = {
		name = mercenary_mutiny_test.10.b
		# AI 拒绝（危机深重时更倾向拒绝）

	type = country_event
	title = mercenary_mutiny_test.11.title
	desc = mercenary_mutiny_test.11.desc
	outcome = negative

	immediate = {
		hidden_effect = {
			# 定位宣布独立的军团国：优先用宗主保存的军团引用（与 test.10 同款，
			# random_subject + has_variable 组合实测不可靠——旧写法导致

			if = {
				limit = {
					exists = var:mercenary_mutiny_company_ref
				}
				var:mercenary_mutiny_company_ref = {
					save_scope_as = mercenary_mutiny_company_scope
				}
			}
			if = {
				limit = {
					NOT = { exists = scope:mercenary_mutiny_company_scope }
				}
				random_subject = {
					limit = {
						has_variable = mercenary_mutiny_disloyal_subject
					}
					save_scope_as = mercenary_mutiny_company_scope
				}
			}
			if = {
				limit = {
					exists = scope:mercenary_mutiny_company_scope
				}
				# 宗主解除与军团国的臣属关系（原版语义：宗主作用域执行，附庸作参数）
				scope:mercenary_mutiny_company_scope = {
					overlord ?= {
						cancel_subject = prev
					}
				}
			}
			if = {
				limit = {
					NOT = { has_variable = blades_and_thrones_military_crisis }
				}
				set_variable = {
					name = blades_and_thrones_military_crisis
					value = 0
				}
			}
			change_variable = {
				name = blades_and_thrones_military_crisis
				add = 20
				min = 0
				max = 100
			}
			set_variable = {
				name = blades_and_thrones_military_crisis_display
				value = var:blades_and_thrones_military_crisis
			}
		}
	}

	option = {
		name = mercenary_mutiny_test.11.a
	}

	option = {
		name = mercenary_mutiny_test.11.b
		high_risk_option = yes
		trigger = {
			exists = scope:mercenary_mutiny_company_scope
		}
		custom_tooltip = mercenary_mutiny_test.11.a.tt
		change_player = scope:mercenary_mutiny_company_scope
	}
