blades_and_thrones_war.1 = { #战争胜利：攻防分开结算的凯旋反馈
	type = country_event
	title = blades_and_thrones_war.1.title
	desc = blades_and_thrones_war.1.desc
	outcome = neutral
	trigger = {
		is_human = yes
	}

	immediate = {
		hidden_effect = {
			# 存入本次犒赏金额（3个月军饷），供选项 tooltip 动态显示
			set_variable = {
				name = blades_and_thrones_triple_pay_display
				value = blades_and_thrones_triple_monthly_pay_cost
			}
		}
	}

	option = {
		name = blades_and_thrones_war.1.a
		trigger = {
			gold >= {
				value = root.blades_and_thrones_triple_monthly_pay_cost
			}
		}
		add_gold = {
			value = root.blades_and_thrones_triple_monthly_pay_cost
			multiply = -1
		}
		"estate(estate_type:zzz_rebalanced_nepal_mercenary_soldiers_estate)" = {
			estate_add_gold = {
				value = root.blades_and_thrones_triple_monthly_pay_cost
			}
		}
		add_estate_satisfaction = {
			type = estate_type:zzz_rebalanced_nepal_mercenary_soldiers_estate
			value = estate_satisfaction_mild_bonus
		}
		if = {
			limit = {
				has_variable = blades_and_thrones_junta_stability
			}
			change_variable = {
				name = blades_and_thrones_junta_stability
				add = 4
				min = 0
				max = 100
			}
		}
		custom_tooltip = blades_and_thrones_war.1.a.tt
	}

	option = {
		name = blades_and_thrones_war.1.b
		add_legitimacy = legitimacy_weak_penalty
		add_estate_satisfaction = {
			type = estate_type:zzz_rebalanced_nepal_mercenary_soldiers_estate
			value = estate_satisfaction_mild_bonus
		}
		if = {
			limit = {
				has_variable = blades_and_thrones_junta_stability
			}
			change_variable = {
				name = blades_and_thrones_junta_stability
				add = 6
				min = 0
				max = 100
			}
		}
		custom_tooltip = blades_and_thrones_war.1.b.tt
	}

	option = {
		name = blades_and_thrones_war.1.c
		add_legitimacy = legitimacy_mild_bonus
		add_stability = stability_weak_bonus
		add_estate_satisfaction = {
			type = estate_type:zzz_rebalanced_nepal_mercenary_soldiers_estate
			value = estate_satisfaction_weak_penalty
		}
		if = {
			limit = {
				has_variable = blades_and_thrones_junta_stability
			}
			change_variable = {
				name = blades_and_thrones_junta_stability
				add = -3
				min = 0
				max = 100
			}
		}
		custom_tooltip = blades_and_thrones_war.1.c.tt
	}

	option = {
		name = blades_and_thrones_war.1.d
		add_prestige = prestige_weak_bonus
