# 但允许驻在国通过“驱逐无地国”CB 将其赶走。
blades_and_thrones_landless_passage = {
	type = diplomacy
	relation_type = oneway
	break_on_war = yes
	annulled_by_peace_treaty = yes
	military_access = yes
	food_access = yes
	receiving_color = define:NMapColors|DIPLOMACY_MIL_ACCESS_COLOR

	category = {
		offer = CATEGORY_ACCESS_ACTIONS
		request = CATEGORY_ACCESS_ACTIONS
		cancellation = CATEGORY_ACCESS_ACTIONS
		break = CATEGORY_ACCESS_ACTIONS
	}

	# 只由 mod 自动创建，不提供外交动作
	offer_visible = { always = no }
	request_visible = { always = no }
	break_visible = { always = no }
	cancel_visible = { always = no }

	wants_to_keep = {
		if = {
			limit = {
				scope:actor = {
					giving_scripted_relation = {
						type = relation_type:blades_and_thrones_landless_passage
						target = scope:recipient
					}
				}
			}
			add = 100
		}
		else = {
			add = -100
		}
	}
}
