blades_and_thrones_military_coup = {
	image = "gfx/interface/illustrations/disaster/coup_attempt.dds"
	monthly_spawn_chance = blades_and_thrones_military_coup_spawn_chance

	can_start = {
		country_has_estate = estate_type:zzz_rebalanced_nepal_mercenary_soldiers_estate
		# 2024-08 需求：只有接纳"职业军队"思潮（professional_armies，文艺复兴
		# 时代）后，军队才可能演变成有组织、有自我意识的夺权力量——中世纪早期
		# 的封建征召兵谈不上"军事政变"。游戏早期（1337+）政变被时代门槛挡住。
		has_embraced_institution = institution:professional_armies
		OR = {
			blades_and_thrones_army_loyalty < 0.15
			# 因果链衔接：哗变被强硬镇压后军人记恨（blades_and_thrones_army_grudge，
			# 24 个月）——镇压只会让政变来得更快，忠诚阈值从 0.15 放宽到 0.25
			AND = {
				has_variable = blades_and_thrones_army_grudge
				blades_and_thrones_army_loyalty < 0.25
			}
			blades_and_thrones_force_coup_flag = yes
		}
		has_any_active_disaster = no
		in_civil_war = no
		NOT = { blades_and_thrones_coup_cooldown_active = yes }
		NOT = { blades_and_thrones_coup_resolved = yes }
		# 2024-08 重构（A+B）：军政府不再是灾难——loyalty.4 抉择前（弹窗间隙）
		# 国家持有军政府改革，防政变灾难在此时再次触发
		NOT = { has_reform = government_reform:blades_and_thrones_military_junta_reform }
	}

	can_end = {
		blades_and_thrones_coup_resolved = yes
	}

	on_start = {
		remove_variable = blades_and_thrones_force_coup
		set_variable = {
			name = blades_and_thrones_coup_start_loyalty
			value = blades_and_thrones_army_loyalty
		}
		set_variable = {
			name = blades_and_thrones_coup_cooldown
			value = yes
			months = 120
		}
		trigger_event_non_silently = {
			id = blades_and_thrones_loyalty.3
		}
	}
}
