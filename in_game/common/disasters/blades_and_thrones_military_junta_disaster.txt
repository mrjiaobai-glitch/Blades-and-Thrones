blades_and_thrones_military_junta_disaster = {
	image = "gfx/interface/illustrations/disaster/coup_attempt.dds"
	monthly_spawn_chance = blades_and_thrones_military_junta_spawn_chance

	can_start = {
		has_reform = government_reform:blades_and_thrones_military_junta_reform
		has_any_active_disaster = no
		NOT = { blades_and_thrones_junta_cooldown_active = yes }
		NOT = { blades_and_thrones_junta_resolved = yes }
	}

	can_end = {
		OR = {
			blades_and_thrones_junta_resolved = yes
			NOT = { has_reform = government_reform:blades_and_thrones_military_junta_reform }
		}
	}

	modifier = {
		stability_cost_efficiency = -0.2
	}

	on_start = {
		set_variable = {
			name = blades_and_thrones_junta_stability
			value = 50
		}
		set_variable = {
			name = blades_and_thrones_junta_stability_growth
			value = 0
		}
		set_variable = {
			name = blades_and_thrones_junta_change_stability_part
			value = blades_and_thrones_junta_change_stability_part
		}
		set_variable = {
			name = blades_and_thrones_junta_change_legitimacy_part
			value = blades_and_thrones_junta_change_legitimacy_part
		}
		set_variable = {
			name = blades_and_thrones_junta_change_loyalty_part
			value = blades_and_thrones_junta_change_loyalty_part
		}
		set_variable = {
			name = blades_and_thrones_junta_change_war_part
			value = blades_and_thrones_junta_change_war_part
		}
	}

	on_monthly = {
		hidden_effect = {
			if = {
				limit = {
					NOT = { has_variable = blades_and_thrones_junta_stability }
				}
				set_variable = {
					name = blades_and_thrones_junta_stability
					value = 50
				}
			}
			if = {
				limit = {
					NOT = { has_variable = blades_and_thrones_junta_stability_growth }
				}
				set_variable = {
					name = blades_and_thrones_junta_stability_growth
					value = 0
				}
			}
			set_variable = {
				name = blades_and_thrones_junta_change_stability_part
				value = blades_and_thrones_junta_change_stability_part
			}
			set_variable = {
				name = blades_and_thrones_junta_change_legitimacy_part
				value = blades_and_thrones_junta_change_legitimacy_part
			}
			set_variable = {
				name = blades_and_thrones_junta_change_loyalty_part
				value = blades_and_thrones_junta_change_loyalty_part
			}
			set_variable = {
				name = blades_and_thrones_junta_change_war_part
				value = blades_and_thrones_junta_change_war_part
			}
			set_variable = {
				name = blades_and_thrones_junta_stability_growth
				value = blades_and_thrones_junta_stability_change
			}
			change_variable = {
				name = blades_and_thrones_junta_stability
				add = var:blades_and_thrones_junta_stability_growth
			}
			if = {
				limit = {
					has_variable = blades_and_thrones_junta_stability
					var:blades_and_thrones_junta_stability > 100
				}
				set_variable = {
					name = blades_and_thrones_junta_stability
					value = 100
				}
			}
			if = {
				limit = {
					has_variable = blades_and_thrones_junta_stability
					var:blades_and_thrones_junta_stability < 0
				}
				set_variable = {
					name = blades_and_thrones_junta_stability
					value = 0
				}
			}
			if = {
				limit = {
					has_variable = blades_and_thrones_junta_stability
					var:blades_and_thrones_junta_stability >= 75
					NOT = { has_variable = blades_and_thrones_junta_consolidated }
					NOT = { has_variable = blades_and_thrones_junta_consolidation_cooldown }
				}
				set_variable = {
					name = blades_and_thrones_junta_consolidation_cooldown
					value = yes
					months = 24
				}
				trigger_event_non_silently = blades_and_thrones_loyalty.4
			}
			if = {
				limit = {
					has_variable = blades_and_thrones_junta_stability
					var:blades_and_thrones_junta_stability >= 75
					has_variable = blades_and_thrones_junta_consolidated
					NOT = { has_variable = blades_and_thrones_junta_consolidation_cooldown }
					NOT = { has_variable = blades_and_thrones_junta_restoration_cooldown }
					NOT = { has_variable = blades_and_thrones_junta_resolved }
				}
				set_variable = {
					name = blades_and_thrones_junta_restoration_cooldown
					value = yes
					months = 36
				}
				trigger_event_non_silently = blades_and_thrones_loyalty.7
			}
			if = {
				limit = {
					has_variable = blades_and_thrones_junta_stability
					var:blades_and_thrones_junta_stability <= 25
					NOT = { has_variable = blades_and_thrones_junta_crisis_cooldown }
					NOT = { has_variable = blades_and_thrones_junta_resolved }
				}
				set_variable = {
					name = blades_and_thrones_junta_crisis_cooldown
					value = yes
					months = 36
				}
				trigger_event_non_silently = blades_and_thrones_loyalty.5
			}
			if = {
				limit = {
					has_variable = blades_and_thrones_junta_stability
					var:blades_and_thrones_junta_stability <= 45
					has_variable = blades_and_thrones_junta_consolidated
					legitimacy < 30
					NOT = { has_variable = blades_and_thrones_junta_restoration_cooldown }
					NOT = { has_variable = blades_and_thrones_junta_resolved }
				}
				set_variable = {
					name = blades_and_thrones_junta_restoration_cooldown
					value = yes
					months = 36
				}
				trigger_event_non_silently = blades_and_thrones_loyalty.6
			}
			if = {
				limit = {
					NOT = { has_variable = blades_and_thrones_junta_decision_cooldown }
					NOT = { has_variable = blades_and_thrones_junta_resolved }
				}
				random_list = {
					15 = {
						set_variable = {
							name = blades_and_thrones_junta_decision_cooldown
							value = yes
							months = 12
						}
						trigger_event_non_silently = blades_and_thrones_junta.1
					}
					85 = { }
				}
			}
			if = {
				limit = {
					has_variable = blades_and_thrones_junta_stability
					var:blades_and_thrones_junta_stability > 45
					var:blades_and_thrones_junta_stability < 75
					NOT = { has_variable = blades_and_thrones_junta_faction_cooldown }
					NOT = { has_variable = blades_and_thrones_junta_resolved }
				}
				random_list = {
					20 = {
						set_variable = {
							name = blades_and_thrones_junta_faction_cooldown
							value = yes
							months = 18
						}
						trigger_event_non_silently = blades_and_thrones_junta.2
					}
					80 = { }
				}
			}
		}
	}

	on_end = {
		if = {
			limit = {
				NOT = { has_variable = blades_and_thrones_junta_forced_restoration_reopen }
			}
			set_variable = {
				name = blades_and_thrones_junta_cooldown
				value = yes
				months = 120
			}
		}
		remove_variable = blades_and_thrones_junta_forced_restoration_reopen
		remove_variable = blades_and_thrones_junta_stability
		remove_variable = blades_and_thrones_junta_stability_growth
		remove_variable = blades_and_thrones_junta_change_stability_part
		remove_variable = blades_and_thrones_junta_change_legitimacy_part
		remove_variable = blades_and_thrones_junta_change_loyalty_part
		remove_variable = blades_and_thrones_junta_change_war_part
		remove_variable = blades_and_thrones_junta_consolidated
		remove_variable = blades_and_thrones_junta_consolidation_cooldown
		remove_variable = blades_and_thrones_junta_crisis_cooldown
		remove_variable = blades_and_thrones_junta_restoration_cooldown
		remove_variable = blades_and_thrones_junta_decision_cooldown
		remove_variable = blades_and_thrones_junta_faction_cooldown
		# 注意：junta_resolved 必须保留——它是 can_start 的防重复标记
		# （NOT { blades_and_thrones_junta_resolved = yes }）。旧代码在 on_end
		# 里把它删掉，导致灾难可无限重复触发，每次解决路径都能重复领取永久
		# 修正奖励（实测那不勒斯拿了 3 个：制度化/清洗/还政各一次）。
		# 强制还政（loyalty.6.a）的重开设计走 coup 链（删 coup_resolved +
		# 36 个月政变冷却），不依赖删除 junta_resolved。
		remove_variable = blades_and_thrones_junta_restoration_refused
	}
}
