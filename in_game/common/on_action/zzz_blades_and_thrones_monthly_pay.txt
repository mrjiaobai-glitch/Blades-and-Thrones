blades_and_thrones_food_pulse = {
	trigger = {
		has_reform = government_reform:blades_and_thrones_food_access_reform
	}
	effect = {
		# —�?通行：四海为家，处处可过�?—�?
		every_country = {
			limit = {
				# 注意：排除自身必须用 this != root（原版标准写法）�?
				# NOT = { this = root } �?on_action �?every 迭代里无效，
				# 会把关系建给自己（create_relation �?"the 2 countries are the same"�?
				this != root
				within_diplomatic_range = root
				NOT = { is_at_war_with = root }
				NOT = {
					root = {
						receives_military_access_from = prev
					}
				}
			}
			# first 必须�?this（迭代元素）而非 prev——实�?on_action �?every_country
			# 主体�?prev 指向外层 root，first = prev 会导�?first = second = root�?
			# create_relation �?"the 2 countries are the same"（原�?create_alliance
			# 同款：first = this）�?
			create_relation = {
				type = relation_type:blades_and_thrones_landless_passage
				first = this
				second = root
				years = 5
			}
		}
		# —�?购粮：驻留之地即有粮 —�?
		every_army = {
			unit_location ?= {
				owner ?= {
					# 排除自身同样�?this != root（owner 作用域内 this = 地主�?
					if = {
						limit = {
							this != root
							NOT = { gives_food_access_to = root }
						}
						save_scope_as = blades_and_thrones_food_owner
					}
				}
			}
			scope:blades_and_thrones_food_owner ?= {
				create_relation = {
					first = scope:blades_and_thrones_food_owner
					second = root
					type = relation_type:food_access
					months = 120
				}
			}
		}
	}
}
