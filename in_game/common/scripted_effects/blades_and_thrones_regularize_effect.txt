		limit = { sub_unit_type = unit_type:n_pinnace }
		scope:mercenary_mutiny_capital = { create_sub_unit_with_owner = { type = n_pinnace owner = scope:mercenary_mutiny_army_country } }
	}
	else = {
		scope:mercenary_mutiny_capital = { create_sub_unit_with_owner = { type = a_footmen owner = scope:mercenary_mutiny_army_country } }
	}
}
