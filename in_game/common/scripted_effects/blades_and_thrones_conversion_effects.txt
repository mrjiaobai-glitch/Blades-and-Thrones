# --- 收编军团（玩家路径）：与玩家交互的收编流程，检查并扣除赎买费用后调用核心效果 ---
# 前置：调用方须先保存具名 scope = scope:blades_and_thrones_company_actor；目标在 scope:target
blades_and_thrones_integrate_mercenary_effect = {
	# 防御性重存：effect 内部拿不到外层的 scope:actor（具名 scope 不跨嵌套 effect 调用）
	# 防御：具名 scope 在嵌套 effect 调用（scope:target = { effect = yes }）内
	# 不可见（实测报 "Scoped object is not valid"）——开头重新从 scope:actor 保存
	scope:actor = {
		save_scope_as = blades_and_thrones_company_actor
	}
	scope:blades_and_thrones_company_actor = {
		set_variable = {
			name = blades_and_thrones_integrate_strength_var
			value = 0
		}
	}

		scope:blades_and_thrones_company_actor = {
			change_variable = {
				name = blades_and_thrones_integrate_strength_var
				add = prev.subunit_strength
			}
		}
	}
	# 组建费用 = 军团总兵力 × 费率（blades_and_thrones_found_company_cost），低于最低价按最低价
	scope:blades_and_thrones_company_actor = {

				value = scope:blades_and_thrones_company_actor.var:blades_and_thrones_integrate_strength_var
				multiply = 1
				min = 100
			}
		}
		if = {
			limit = { scope:blades_and_thrones_company_actor.var:blades_and_thrones_integrate_cost_var > scope:blades_and_thrones_company_actor.gold }
			set_variable = {
				name = blades_and_thrones_integrate_cost_var
				value = scope:blades_and_thrones_company_actor.gold
			}
		}
		add_gold = {
			value = scope:blades_and_thrones_company_actor.var:blades_and_thrones_integrate_cost_var
			multiply = -1
		}
	}
	blades_and_thrones_integrate_mercenary_core_effect = yes
	# 收编反馈：军事危机 -20（先兜底初始化再 change_variable，并同步 _display 显示变量）
	scope:blades_and_thrones_company_actor = {

			set_variable = {
				name = blades_and_thrones_military_crisis
				value = 0
			}
		}
		change_variable = {
			name = blades_and_thrones_military_crisis
			add = -20
			min = 0
			max = 100
		}
		set_variable = {
			name = blades_and_thrones_military_crisis_display
			value = scope:blades_and_thrones_company_actor.var:blades_and_thrones_military_crisis
		}
	}
}

	scope:blades_and_thrones_company_actor = {

	unit_location ?= {

# --- 收编自立军团（灾变和解后的不忠附庸）：赎买归队，宗主重新拿回部队控制权 ---

# 区分两条路径：volunteer 标记（组建/外派而来的军团，免费归队）与本 effect 的赎买路径（按兵力 × 100 金/千人付费）

		limit = {
			has_variable = mercenary_mutiny_exiles_country

	remove_country_modifier = mercenary_mutiny_placated
	remove_variable = mercenary_mutiny_disloyal_subject

	scope:actor = {
		remove_variable = mercenary_mutiny_company_ref
	}

# --- 遣散军队：把部队就地解散并折算成士兵 POP（复员为阶层人口，corps 转 pop）---
# 前置：本 effect 的作用域 = 目标单位（由调用方直接 scope:target 进来）
	# 注意：add_pop 的 size 不能依赖 actor 预置变量（UI 预览不执行 set_variable）；
	# 若 add_pop 处需要预置变量（本 effect 不需要——size 直接用部队兵力换算），
	# 则应改为在 sub_unit 层直接换算，避免预览与实际执行不一致（POP 归属落点 owner）
blades_and_thrones_demobilize_army_effect = {
	unit_location ?= {
		save_scope_as = blades_and_thrones_demob_location
	}
	if = {
		limit = { exists = scope:blades_and_thrones_demob_location }
		every_sub_unit = {
			save_scope_as = blades_and_thrones_current_subunit
			scope:blades_and_thrones_demob_location = {
				add_pop = {
					# root 即本 effect 作用域（目标单位），不依赖调用方具名 scope（心得#31 坑）
					culture = root.owner.culture
					religion = root.owner.religion
					type = pop_type:soldiers
					size = {
						value = scope:blades_and_thrones_current_subunit.subunit_strength
						multiply = mercenary_mutiny_pop_conversion_ratio
						min = 0.001
					}
				}
			}
		}
	}
	destroy_unit = yes
}
