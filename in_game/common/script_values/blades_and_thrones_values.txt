# 军事危机值月度变化量（0-100 的危机条）
# 佣兵威胁比率越高涨得越快；低忠诚/低满意度/战时加剧；军队健康时消退。
blades_and_thrones_military_crisis_change = {
	value = 0
	if = {
		limit = { mercenary_mutiny_threat_ratio >= 4 }
		add = 4.0
	}
	else_if = {
		limit = { mercenary_mutiny_threat_ratio >= 3 }
		add = 3.0
	}
	else_if = {
		limit = { mercenary_mutiny_threat_ratio >= 2 }
		add = 2.0
	}
	else_if = {
		limit = { mercenary_mutiny_threat_ratio >= 1.5 }
		add = 1.5
	}
	else_if = {
		limit = { mercenary_mutiny_threat_ratio >= 1 }
		add = 1.0
	}
	if = {
		limit = { blades_and_thrones_army_loyalty < 0.35 }
		add = 0.5
	}
	if = {
		limit = { blades_and_thrones_army_loyalty < 0.2 }
		add = 0.5
	}
	if = {
		limit = {
			country_has_estate = estate_type:zzz_rebalanced_nepal_mercenary_soldiers_estate
			estate_satisfaction:zzz_rebalanced_nepal_mercenary_soldiers_estate < 0.5
		}
		add = 0.25
	}
	if = {
		limit = { at_war = yes }
		add = 0.25
	}
	# 军队健康时危机消退
	if = {
		limit = { mercenary_mutiny_threat_ratio < 0.75 }
		subtract = 1.5
	}
	else_if = {
		limit = { mercenary_mutiny_threat_ratio < 1 }
		subtract = 0.75
	}
	# 时代阶段化：早期佣兵依赖是常态（威胁权重放宽），晚期更危险（危机积累更快）
	# 早期（传统/文艺复兴/地理大发现）：消退更快（军队健康时危机压力小）
	if = {
		limit = {
			OR = {
				current_age = age_1_traditions
				current_age = age_2_renaissance
				current_age = age_3_discovery
			}
		}
		subtract = 0.3
	}
	# 晚期（专制/革命）：佣兵依赖是时代错位——危机积累更快
	if = {
		limit = {
			OR = {
				current_age = age_5_absolutism
				current_age = age_6_revolutions
			}
		}
		add = 0.5
	}
}
